using AgroGuia.Aplicacion.DTO.DTOS.Auth;
using AgroGuia.Aplicacion.Servicio;
using AgroGuia.Dominio.Modelo.Abstracciones;
using AgroGuia.Dominio.Modelo.Entidades;
using Microsoft.Extensions.Logging;

namespace AgroGuia.Aplicacion.ServicioImpl;

public class AuthServicioImpl : IAuthServicio
{
    private const string CredencialesInvalidas = "Credenciales inválidas";

    private readonly IUsuarioRepositorio _usuarioRepositorio;
    private readonly IJwtService _jwtService;
    private readonly ILogger<AuthServicioImpl> _logger;

    public AuthServicioImpl(
        IUsuarioRepositorio usuarioRepositorio,
        IJwtService jwtService,
        ILogger<AuthServicioImpl> logger)
    {
        _usuarioRepositorio = usuarioRepositorio
            ?? throw new ArgumentNullException(nameof(usuarioRepositorio));
        _jwtService = jwtService
            ?? throw new ArgumentNullException(nameof(jwtService));
        _logger = logger
            ?? throw new ArgumentNullException(nameof(logger));
    }

    public async Task<LoginResponseDto> RegistrarAsync(RegistroRequestDto request)
    {
        try
        {
            if (string.IsNullOrWhiteSpace(request.NombreCompleto) ||
                string.IsNullOrWhiteSpace(request.Email) ||
                string.IsNullOrWhiteSpace(request.Password) ||
                request.Password.Length < 6)
                return Falla("Datos de registro incompletos o contraseña demasiado corta");

            var existeUsuario = await _usuarioRepositorio.ObtenerPorEmailAsync(request.Email);
            if (existeUsuario != null)
                return Falla("El email ya está registrado");

            var usuario = new Usuarios
            {
                NombreCompleto = request.NombreCompleto.Trim(),
                Email = request.Email.Trim(),
                PasswordHash = BCrypt.Net.BCrypt.HashPassword(request.Password),
                Telefono = request.Telefono,
                Rol = "Agricultor",              // el rol nunca viene del cliente
                FechaRegistro = DateTime.UtcNow,
                Activo = true
            };

            await _usuarioRepositorio.CrearUsuarioAsync(usuario);

            return await EmitirSesionAsync(usuario, "Usuario registrado correctamente");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error en el registro de usuario");
            return Falla("No se pudo completar el registro");
        }
    }

    public async Task<LoginResponseDto> LoginAsync(LoginRequestDto request)
    {
        try
        {
            var usuario = await _usuarioRepositorio.ObtenerPorEmailAsync(request.Email);

            // Mismo mensaje en ambos casos: no revela qué correos existen
            if (usuario == null || !usuario.Activo)
                return Falla(CredencialesInvalidas);

            bool passwordCorrecta = BCrypt.Net.BCrypt.Verify(request.Password, usuario.PasswordHash);
            if (!passwordCorrecta)
                return Falla(CredencialesInvalidas);

            usuario.UltimoAcceso = DateTime.UtcNow;   // se guarda dentro de EmitirSesionAsync

            return await EmitirSesionAsync(usuario, "Login exitoso");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error en el login");
            return Falla("No se pudo iniciar sesión");
        }
    }

    public async Task<LoginResponseDto> RefrescarAsync(string refreshToken)
    {
        try
        {
            if (string.IsNullOrWhiteSpace(refreshToken))
                return Falla("Sesión expirada");

            var hash = _jwtService.HashRefreshToken(refreshToken);
            var usuario = await _usuarioRepositorio.ObtenerPorRefreshTokenAsync(hash);

            if (usuario == null ||
                usuario.RefreshTokenExpiry == null ||
                usuario.RefreshTokenExpiry < DateTime.UtcNow ||
                !usuario.Activo)
                return Falla("Sesión expirada");

            // Rotación: cada refresh emite uno nuevo y el anterior deja de servir
            return await EmitirSesionAsync(usuario, "Token renovado");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error al renovar el token");
            return Falla("No se pudo renovar la sesión");
        }
    }

    public async Task CerrarSesionAsync(long usuarioId)
    {
        var usuario = await _usuarioRepositorio.ObtenerPorIdAsync(usuarioId);
        if (usuario == null) return;

        usuario.RefreshToken = null;
        usuario.RefreshTokenExpiry = null;
        await _usuarioRepositorio.ActualizarUsuarioAsync(usuario);
    }

    // Genera access token + refresh token, guarda el hash del refresh y devuelve la sesión
    private async Task<LoginResponseDto> EmitirSesionAsync(Usuarios usuario, string mensaje)
    {
        string token = _jwtService.GenerarToken(
            usuario.Id, usuario.NombreCompleto, usuario.Email, usuario.Rol);

        string refreshToken = _jwtService.GenerarRefreshToken();

        usuario.RefreshToken = _jwtService.HashRefreshToken(refreshToken);
        usuario.RefreshTokenExpiry = DateTime.UtcNow.AddDays(_jwtService.DiasRefreshToken);
        await _usuarioRepositorio.ActualizarUsuarioAsync(usuario);

        return new LoginResponseDto
        {
            Exito = true,
            Mensaje = mensaje,
            UsuarioId = usuario.Id,
            NombreCompleto = usuario.NombreCompleto,
            Email = usuario.Email,
            Rol = usuario.Rol,
            Token = token,
            RefreshToken = refreshToken,
            ExpiraEn = DateTime.UtcNow.AddMinutes(_jwtService.MinutosExpiracion)
        };
    }

    private static LoginResponseDto Falla(string mensaje) =>
        new LoginResponseDto { Exito = false, Mensaje = mensaje };
}

public class ChatServicioImpl : IChatServicio
{
    private const int MaxTurnosHistorial = 8; // 8 mensajes ≈ 4 pares user/assistant
    private const int MaxCharsHistorial = 1800;

    private readonly ContextRetriever _contextRetriever;
    private readonly PromptBuilder _promptBuilder;
    private readonly IChatCompletionService _chat;
    private readonly IConversacionRepositorio _conversacionRepositorio;
    private readonly ILogger<ChatServicioImpl> _log;

    // Cultivos frecuentes (amplía según tu catálogo)
    private static readonly string[] CultivosConocidos =
    {
        "quinua", "quinoa", "papa", "maíz", "maiz", "frejol", "fréjol", "frijol",
        "trigo", "cebada", "haba", "arveja", "ulluco", "oca", "mashua", "tomate",
        "cebolla", "ajo", "lenteja", "chocho", "amaranto"

    public ChatServicioImpl(
        ContextRetriever contextRetriever,

        IChatCompletionService chat,

        ILogger<ChatServicioImpl> log)

    public async Task<ChatResponseDto> ProcesarConsultaAsync(ConsultaRequestDto request)
    {
        try
        {
            if (string.IsNullOrWhiteSpace(request.Consulta))
            {
                return new ChatResponseDto
                {
                    Exito = false,
                    Respuesta = "Escribe una consulta agronómica.",
                    FechaRespuesta = DateTime.UtcNow
                };
            }

            // ----- 0) Historial de ESTA conversación -----
            var previos = await CargarMensajesPreviosAsync(request.ConversacionId);
            var historialTexto = FormatearHistorial(previos);
            var consultaRag = AmpliarConsultaParaRag(request.Consulta, previos);

            _log.LogInformation(
                "Hilo conv={Id}: historial={N} msgs | ragQuery=\"{Q}\"",
                request.ConversacionId, previos.Count, consultaRag);

            // ----- 1) RAG con consulta ampliada (mantiene cultivo/tema del hilo) -----
            var reloj = Stopwatch.StartNew();
            var contextos = await _contextRetriever.ObtenerContextoAsync(consultaRag);
            var msContexto = reloj.ElapsedMilliseconds;

            if (contextos.Count == 0)
            {
                var sinInfo = PromptBuilder.RespuestaSinInformacion;
                await GuardarParAsync(request, sinInfo, 0);
                return Ok(sinInfo, 0);
            }

            // ----- 2) Prompt = historial + contexto + pregunta actual -----
            var usados = _promptBuilder.SeleccionarContexto(contextos);
            var basePrompt = _promptBuilder.Construir(request.Consulta.Trim(), usados, historialTexto);

            var prompt = new ChatPrompt(
                basePrompt.Sistema,
                basePrompt.Usuario,
                NormalizarModo(request.ModoModelo));

            // ----- 3) LLM -----
            reloj.Restart();
            var result = await _chat.GenerarAsync(prompt);

            _log.LogInformation(
                "Chat: contexto {MsCtx} ms ({Usados}/{Recuperados} frag.) | modelo {MsLlm} ms | modo {Modo}",
                msContexto, usados.Count, contextos.Count,
                reloj.ElapsedMilliseconds, prompt.ModoModelo ?? "(defecto)");

            string texto;
            if (result.Exito && !string.IsNullOrWhiteSpace(result.Respuesta))
            {
                texto = result.Respuesta.Trim();
            }
            else
            {
                texto = string.IsNullOrWhiteSpace(result.ErrorMensaje)
                    ? PromptBuilder.RespuestaSinInformacion
                    : "El asistente no pudo generar la respuesta. Intenta de nuevo o usa el modo Rápido.";
            }

            var tokens = result.TokensEntrada + result.TokensSalida;
            if (result.Exito)
                await GuardarParAsync(request, texto, tokens);

            // Opcional: persistir cultivo del hilo si la entidad/repo lo soportan
            await IntentarActualizarCultivoHiloAsync(request.ConversacionId, request.Consulta, previos);

            return Ok(texto, tokens, result.Exito ? null : result.ErrorMensaje);
        }
        catch (Exception ex)
        {
            _log.LogError(ex, "Error en ChatServicio");
            return new ChatResponseDto
            {
                Exito = false,
                Respuesta = "Lo siento, ocurrió un error interno al procesar tu consulta.",
                TokensUsados = 0,
                FechaRespuesta = DateTime.UtcNow,
                ErrorMensaje = ex.Message
            };
        }
    }

    private async Task<IReadOnlyList<Mensajes>> CargarMensajesPreviosAsync(long conversacionId)
    {
        if (conversacionId <= 0)
            return Array.Empty<Mensajes>();

        var list = await _conversacionRepositorio.ListarMensajesAsync(conversacionId, take: 50);

        return list
            .OrderBy(m => m.Fecha)
            .TakeLast(MaxTurnosHistorial)
            .ToList();
    }
    private static string? FormatearHistorial(IReadOnlyList<Mensajes> previos)
    {
        if (previos.Count == 0) return null;

        var sb = new StringBuilder();
        foreach (var m in previos)
        {
            var rol = string.Equals(m.Rol, "user", StringComparison.OrdinalIgnoreCase)
                ? "Usuario"
                : "Asistente";
            var contenido = (m.Contenido ?? "").Trim();
            if (contenido.Length > 400)
                contenido = contenido[..400].TrimEnd() + "…";
            sb.Append(rol).Append(": ").Append(contenido).Append('\n');
        }

        var texto = sb.ToString();
        if (texto.Length > MaxCharsHistorial)
            texto = texto[^MaxCharsHistorial..]; // cola del hilo (lo más reciente)

        return texto;
    }

    /// <summary>
    /// Si la pregunta es corta / seguimiento, añade cultivo y pistas del historial
    /// para que el RAG no busque solo "más métodos".
    /// </summary>
    private static string AmpliarConsultaParaRag(string preguntaActual, IReadOnlyList<Mensajes> previos)
    {
        var q = preguntaActual.Trim();
        if (previos.Count == 0) return q;

        var cultivoEnPregunta = DetectarCultivo(q);
        var cultivoHilo = cultivoEnPregunta
            ?? DetectarCultivoEnMensajes(previos)
            ?? "";

        var esSeguimiento = PareceSeguimiento(q) || cultivoEnPregunta is null;

        if (!esSeguimiento && cultivoEnPregunta is not null)
            return q; // pregunta nueva ya explícita

        // Últimas preguntas del usuario (pistas: trips, jabón, siembra…)
        var pistas = previos
            .Where(m => string.Equals(m.Rol, "user", StringComparison.OrdinalIgnoreCase))
            .Select(m => m.Contenido ?? "")
            .Reverse()
            .Take(3)
            .ToList();

        var extra = new StringBuilder(q);
        if (!string.IsNullOrEmpty(cultivoHilo) &&
            q.IndexOf(cultivoHilo, StringComparison.OrdinalIgnoreCase) < 0)
        {
            extra.Append(' ').Append(cultivoHilo);
        }

        // Añade 1–2 frases previas del user si el mensaje actual es muy corto
        if (q.Length < 80 || PareceSeguimiento(q))
        {
            foreach (var p in pistas)
            {
                if (string.IsNullOrWhiteSpace(p)) continue;
                if (p.Equals(q, StringComparison.OrdinalIgnoreCase)) continue;
                extra.Append(' ').Append(p.Length > 120 ? p[..120] : p);
                break; // una pista basta para no ensuciar el embed
            }
        }

        return Regex.Replace(extra.ToString(), @"\s+", " ").Trim();
    }

    private static bool PareceSeguimiento(string q)
    {
        var t = q.Trim().ToLowerInvariant();
        if (t.Length <= 48) return true;
        return t.StartsWith("y ") || t.StartsWith("y el") || t.StartsWith("y la")
            || t.StartsWith("más ") || t.StartsWith("mas ")
            || t.Contains("otro método") || t.Contains("otros metodos")
            || t.Contains("más métodos") || t.Contains("mas metodos")
            || t.StartsWith("eso") || t.StartsWith("y si")
            || t.Contains("también") || t.Contains("tambien");
    }

    private static string? DetectarCultivo(string texto)
    {
        var t = texto.ToLowerInvariant();
        foreach (var c in CultivosConocidos)
            if (t.Contains(c)) return c is "quinoa" ? "quinua" : c is "maiz" ? "maíz" : c;
        return null;
    }

    private static string? DetectarCultivoEnMensajes(IReadOnlyList<Mensajes> msgs)
    {
        for (int i = msgs.Count - 1; i >= 0; i--)
        {
            if (!string.Equals(msgs[i].Rol, "user", StringComparison.OrdinalIgnoreCase))
                continue;
            var c = DetectarCultivo(msgs[i].Contenido ?? "");
            if (c is not null) return c;
        }
        return null;
    }

    private async Task IntentarActualizarCultivoHiloAsync(
    long conversacionId, string pregunta, IReadOnlyList<Mensajes> previos)
    {
        if (conversacionId <= 0) return;

        var cultivo = DetectarCultivo(pregunta) ?? DetectarCultivoEnMensajes(previos);
        if (cultivo is null) return;

        var conv = await _conversacionRepositorio.ObtenerPorIdAsync(conversacionId);
        if (conv is null) return;

        // Ya tiene el mismo cultivo → no hace falta guardar
        if (string.Equals(conv.CultivoActual, cultivo, StringComparison.OrdinalIgnoreCase))
            return;

        conv.CultivoActual = cultivo;

        // Mejora el título solo si sigue siendo el por defecto
        var tituloActual = (conv.Titulo ?? "").Trim();
        if (string.IsNullOrEmpty(tituloActual) ||
            tituloActual.Equals("Nueva consulta", StringComparison.OrdinalIgnoreCase) ||
            tituloActual.Equals("Nueva Consulta Agronómica", StringComparison.OrdinalIgnoreCase))
        {
            // Primera letra en mayúscula
            var nombre = char.ToUpper(cultivo[0]) + cultivo[1..];
            conv.Titulo = $"Consulta sobre {nombre}";
        }

        await _conversacionRepositorio.ActualizarAsync(conv);
    }

    // -------------------------------------------------------------------------

    private static string? NormalizarModo(string? modo)
    {
        if (string.IsNullOrWhiteSpace(modo)) return null;
        modo = modo.Trim().ToLowerInvariant();
        return modo is "rapido" or "preciso" ? modo : null;
    }

    private static ChatResponseDto Ok(string respuesta, int tokens, string? error = null) => new()
    {
        Exito = true,
        Respuesta = respuesta,
        TokensUsados = tokens,
        FechaRespuesta = DateTime.UtcNow,
        ErrorMensaje = error ?? ""
    };

    private async Task GuardarParAsync(ConsultaRequestDto request, string respuesta, int tokens)
    {
        await _conversacionRepositorio.MensajeGuardarAsync(new Mensajes
        {
            ConversacionId = request.ConversacionId,
            Rol = "user",
            Contenido = request.Consulta,
            Fecha = DateTime.UtcNow
        });

        await _conversacionRepositorio.MensajeGuardarAsync(new Mensajes
        {
            ConversacionId = request.ConversacionId,
            Rol = "assistant",
            Contenido = respuesta,
            Fecha = DateTime.UtcNow,
            Tokens = tokens
        });
    }

    public int DiasRefreshToken => LeerEntero("Jwt:RefreshTokenDays", 7);

    public string GenerarToken(long idUsuario, string nombre, string email, string rol)
    {
        var key = Encoding.UTF8.GetBytes(
            _configuration["Jwt:SecretKey"]
            ?? throw new InvalidOperationException("Jwt:SecretKey no configurado"));

        var ahora = DateTime.UtcNow;

        var claims = new List<Claim>
        {
            new Claim("IdUsuario", idUsuario.ToString()),
            new Claim(ClaimTypes.NameIdentifier, idUsuario.ToString()),
            new Claim(ClaimTypes.Name, nombre),
            new Claim(ClaimTypes.Email, email),
            new Claim(ClaimTypes.Role, rol)
        };

        var tokenDescriptor = new SecurityTokenDescriptor
        {
            Subject = new ClaimsIdentity(claims),
            NotBefore = ahora,
            Expires = ahora.AddMinutes(MinutosExpiracion),
            Issuer = _configuration["Jwt:Issuer"],
            Audience = _configuration["Jwt:Audience"],
            SigningCredentials = new SigningCredentials(
                new SymmetricSecurityKey(key),
                SecurityAlgorithms.HmacSha256Signature)
        };

        var tokenHandler = new JwtSecurityTokenHandler();
        return tokenHandler.WriteToken(tokenHandler.CreateToken(tokenDescriptor));
    }

        Convert.ToBase64String(RandomNumberGenerator.GetBytes(64));

            throw new ArgumentException("La imagen está vacía.", nameof(imagen));

        var hash = Convert.ToHexString(SHA256.HashData(imagen)).ToLowerInvariant();
        var nombre = hash + ".jpg";
        var dir = Path.Combine(_raiz, usuarioId.ToString());
        Directory.CreateDirectory(dir);

        var abs = Path.Combine(dir, nombre);
        if (!File.Exists(abs))
            await File.WriteAllBytesAsync(abs, imagen, ct);

        return new FotoDiagnosticoGuardadaDto
        {
            RutaRelativa = $"diagnosticos/{usuarioId}/{nombre}",
            Hash = hash,
            NombreArchivo = nombre
        };
    }

    public async Task<byte[]?> LeerAsync(string rutaRelativa, CancellationToken ct = default)
    {
        var abs = Resolver(rutaRelativa);
        if (abs is null || !File.Exists(abs)) return null;
        return await File.ReadAllBytesAsync(abs, ct);
    }

    private string? Resolver(string rutaRelativa)
    {
        if (string.IsNullOrWhiteSpace(rutaRelativa)) return null;

        var rel = rutaRelativa.Replace('\\', '/').TrimStart('/');
        const string prefijo = "diagnosticos/";
        if (!rel.StartsWith(prefijo, StringComparison.OrdinalIgnoreCase)) return null;

        var abs = Path.GetFullPath(Path.Combine(_raiz, rel[prefijo.Length..]));
        var raiz = _raiz.TrimEnd(Path.DirectorySeparatorChar) + Path.DirectorySeparatorChar;
        return abs.StartsWith(raiz, StringComparison.OrdinalIgnoreCase) ? abs : null;
    }

    public async Task<int> CargarDocumentosDesdeCarpetaAsync(string carpetaPath, CancellationToken ct = default)
    {
        if (!Directory.Exists(carpetaPath))
            throw new DirectoryNotFoundException($"Carpeta no encontrada: {carpetaPath}");

        var pdfs = Directory.GetFiles(carpetaPath, "*.pdf");
        var snapshot = await _index.ObtenerAsync(ct);
        var hashes = snapshot.Chunks.Where(c => c.Hash != null).Select(c => c.Hash!).ToHashSet();
        var archivos = snapshot.Chunks.Select(c => c.Archivo).ToHashSet(StringComparer.OrdinalIgnoreCase);

        _log.LogInformation("Iniciando carga de {N} PDFs", pdfs.Length);
        int cargados = 0;

        foreach (var pdf in pdfs)
        {
            ct.ThrowIfCancellationRequested();
            var archivo = Path.GetFileName(pdf);
            try
            {
                var hash = await HashAsync(pdf, ct);
                if (hashes.Contains(hash)) { _log.LogInformation("Sin cambios, se omite: {A}", archivo); continue; }

                if (archivos.Contains(archivo))
                {
                    _log.LogWarning("{A} cambió pero su versión anterior sigue activa. Desactívala antes: " +
                        "UPDATE dbo.EmbeddingChunks SET Activo = 0 WHERE JSON_VALUE(Metadata,'$.archivo') = '{A}'",
                        archivo, archivo);
                    continue;
                }

                if (await ProcesarPdfAsync(pdf, archivo, hash, ct)) { cargados++; hashes.Add(hash); }
            }
            catch (OperationCanceledException) { throw; }
            catch (Exception ex) { _log.LogError(ex, "Error procesando {A}", archivo); }
        }

        if (cargados > 0) _index.Invalidar();
        _log.LogInformation("Carga completada. Documentos nuevos: {N}", cargados);
        return cargados;
    }

/// <summary>
/// MEJORAS sobre DividirEnChunksInteligente (Substring por caracteres):
///  - Nunca corta a mitad de palabra. En tu tabla se ven chunks que empiezan con
///    "orado a partir de...", "CIÃ“N GENERAL", "osicionÃ¡ndose" y palabras clave como "rganizaciones":

///  - Agrupa por pÃ¡rrafos/bloques, respeta secciones (detecta tÃ­tulos) y guarda pÃ¡gina y secciÃ³n.
///  - Parte pÃ¡rrafos gigantes por oraciones sin cortar "Ing.", "Fig." ni "2.5 kg".
///  - Descarta chunks con poca proporciÃ³n de letras (tablas rotas, Ã­ndices).

///    cada paso reemplazaba al anterior y el texto desaparecÃ­a del contenido Y de la secciÃ³n. Ahora la numeraciÃ³n
///    solo cuenta como tÃ­tulo si tiene subniveles ("2.1 Plagas"); "1. INTRODUCCIÃ“N" sigue siendo tÃ­tulo por ir en mayÃºsculas.
///  - Las lÃ­neas de pura basura ("Â· ~ ^ 1 ; 2 |") se descartan por separado: antes sobrevivÃ­an dentro de un chunk bueno.

    private static readonly Regex Numerado = new(@"^\d{1,2}(\.\d{1,2}){1,3}[.)]?\s+\p{Lu}", RegexOptions.Compiled);
    private static readonly Regex PalabraTitulo = new(@"^(CAP[IÃ]TULO|ANEXO|M[OÃ“]DULO|SECCI[OÃ“]N|UNIDAD|TEMA)\b", RegexOptions.Compiled);
    private static readonly Regex FinOracion = new(@"(?<=[.!?;])\s+(?=[\p{Lu}Â¿Â¡])", RegexOptions.Compiled);
    private static readonly Regex Abreviatura = new(
        @"(?:\b(?:Ing|Dr|Dra|Sr|Sra|Lic|Fig|Cap|Vol|NÃºm|Prof|Msc|Mtr)|\b\p{Lu})\.$", RegexOptions.Compiled);

    private sealed record Unidad(string Texto, int Pagina, string? Seccion);
    private readonly ChunkingOptions _o;
    private readonly ILogger<ParagraphChunker> _log;

    public ParagraphChunker(IOptions<ChunkingOptions> opt, ILogger<ParagraphChunker> log)
    {

    public IReadOnlyList<ChunkPreparado> Dividir(IReadOnlyList<BloqueTexto> bloques)
    {
        // 1) Bloques â†’ unidades (tÃ­tulos solo actualizan la secciÃ³n actual)
        var unidades = new List<Unidad>();
        string? seccion = null;
        int sinLetras = 0, pocoTexto = 0, muyCortos = 0;
        var muestras = new List<string>();

        foreach (var b in bloques)
        {
            if (EsTitulo(b.Texto)) { seccion = b.Texto; continue; }
            foreach (var parte in PartirSiEsLargo(b.Texto))
            {
                if (RatioLetras(parte) < _o.RatioLetrasUnidadMinimo) { sinLetras++; continue; }
                unidades.Add(new Unidad(parte, b.Pagina, seccion));
            }
        }

        // 2) Agrupar unidades hasta MaxChars sin cruzar secciones
        var crudos = new List<(string Texto, string? Seccion, int Pagina)>();
        var actual = new List<Unidad>();
        int len = 0, nuevos = 0;

        foreach (var u in unidades)
        {
            bool cambioSeccion = actual.Count > 0 && actual[^1].Seccion != u.Seccion;
            if (nuevos > 0 && (cambioSeccion || len + u.Texto.Length + 2 > _o.MaxChars))
            {
                crudos.Add(Armar(actual));
                actual = cambioSeccion ? new List<Unidad>() : actual.TakeLast(_o.OverlapParrafos).ToList();
                len = actual.Sum(x => x.Texto.Length + 2);
                nuevos = 0;
            }
            actual.Add(u); len += u.Texto.Length + 2; nuevos++;
        }
        if (nuevos > 0) crudos.Add(Armar(actual));

        // 3) Filtrado de calidad y fusiÃ³n de restos pequeÃ±os
        var finales = new List<(string Texto, string? Seccion, int Pagina)>();
        foreach (var c in crudos)
        {
            if (RatioLetras(c.Texto) < _o.RatioLetrasMinimo)
            {
                pocoTexto++;
                if (muestras.Count < 3) muestras.Add(c.Texto.Length > 80 ? c.Texto[..80] : c.Texto);
                continue;
            }

            if (c.Texto.Length < _o.MinChars)
            {
                if (finales.Count > 0 && finales[^1].Seccion == c.Seccion &&
                    finales[^1].Texto.Length + c.Texto.Length + 2 <= _o.MaxChars * 1.3)
                {
                    var p = finales[^1];
                    finales[^1] = (p.Texto + "\n\n" + c.Texto, p.Seccion, p.Pagina);
                    continue;
                }
                if (c.Texto.Length < 60) { muyCortos++; continue; }
            }
            finales.Add(c);

                "Ejemplos de poco texto: {M}",

        return finales.Select((c, i) => new ChunkPreparado(i, c.Texto, c.Seccion, c.Pagina)).ToList();
    }

    private static (string, string?, int) Armar(List<Unidad> u) =>
        (string.Join("\n\n", u.Select(x => x.Texto)), u[0].Seccion, u[0].Pagina);

    private static double RatioLetras(string s)
    {
        int visibles = 0, letras = 0;
        foreach (var c in s)
        {
            if (char.IsWhiteSpace(c)) continue;
            visibles++;
            if (char.IsLetter(c)) letras++;
        }
        return visibles == 0 ? 0 : (double)letras / visibles;
    }

    private static bool EsTitulo(string t)
    {
        if (t.Length < 3 || t.Length > 100) return false;
        if (t[^1] is '.' or ',' or ';' or ':') return false;
        int palabras = t.Count(c => c == ' ') + 1;

        if (PalabraTitulo.IsMatch(t)) return true;
        if (Numerado.IsMatch(t)) return palabras <= 10;

        int letras = t.Count(char.IsLetter);
        if (letras < 4) return false;
        int mayus = t.Count(char.IsUpper);
        return palabras <= 12 && (double)mayus / letras >= 0.8;
    }

    private IEnumerable<string> PartirSiEsLargo(string texto)

        foreach (var o in oraciones)

                {
                    yield return sb.ToString();
                    sb.Clear();
                }
                if (sb.Length > 0) sb.Append(' ');
                sb.Append(trozo);
            }
        if (sb.Length > 0) yield return sb.ToString();
    }

    private IEnumerable<string> CortarPorPalabras(string s)
    {
        if (s.Length <= _o.MaxChars) { yield return s; yield break; }
        int pos = 0;
        while (pos < s.Length)
        {
            int fin = Math.Min(pos + _o.MaxChars, s.Length);
            if (fin < s.Length)
            {
                int sp = s.LastIndexOf(' ', fin - 1, fin - pos);
                if (sp > pos) fin = sp;
            }
            yield return s.Substring(pos, fin - pos).Trim();
            pos = fin;
            while (pos < s.Length && s[pos] == ' ') pos++;
        }
    }

///  - Elimina lÃ­neas de Ã­ndice ("CapÃ­tulo 1 ........ 12").

public sealed class TextCleaner : ITextCleaner
{
    private static readonly Regex Hyphen = new(@"(?<=\p{L}{2})-\s+(?=\p{Ll})", RegexOptions.Compiled);
    private static readonly Regex Espacios = new(@"\s+", RegexOptions.Compiled);
    private static readonly Regex Puntos = new(@"\.{3,}|(\.\s){4,}", RegexOptions.Compiled);
    private static readonly Regex LineaIndice = new(@"(\.{4,}|(\.\s){4,})\s*\d+\s*$", RegexOptions.Compiled);
    private static readonly Regex SoloNumero = new(@"^\W*\d{1,4}\W*$", RegexOptions.Compiled);
    private static readonly Regex Digitos = new(@"\d+", RegexOptions.Compiled);
    private static readonly Regex Control = new(@"[\u0000-\u0008\u000B\u000C\u000E-\u001F]", RegexOptions.Compiled);

    public IReadOnlyList<BloqueTexto> Limpiar(DocumentoExtraido documento)
    {
        int totalPaginas = documento.Paginas.Count;

        // Firma de bloques cortos â†’ en cuÃ¡ntas pÃ¡ginas distintas aparece.
        var paginasPorFirma = new Dictionary<string, HashSet<int>>();
        foreach (var p in documento.Paginas)
            foreach (var b in p.Bloques.Where(b => b.Length <= 150))
                (paginasPorFirma.TryGetValue(Firma(b), out var set)
                    ? set : paginasPorFirma[Firma(b)] = new HashSet<int>()).Add(p.Numero);

        var ruido = totalPaginas >= 4
            ? paginasPorFirma.Where(kv => (double)kv.Value.Count / totalPaginas >= 0.40)
                             .Select(kv => kv.Key).ToHashSet()
            : new HashSet<string>();

        var resultado = new List<BloqueTexto>();
        foreach (var p in documento.Paginas)
            foreach (var crudo in p.Bloques)
            {
                if (LineaIndice.IsMatch(crudo)) continue;
                if (crudo.Length <= 150 && ruido.Contains(Firma(crudo))) continue;

                var t = Normalizar(crudo);
                if (t.Length == 0 || SoloNumero.IsMatch(t)) continue;
                resultado.Add(new BloqueTexto(p.Numero, t));
            }
        return resultado;
    }

    private static string Firma(string s) =>
        Digitos.Replace(Espacios.Replace(s.Trim().ToLowerInvariant(), " "), "#");

    private static string Normalizar(string s)
    {
        s = s.Normalize(NormalizationForm.FormKC).Replace("\u00AD", "");
        s = Control.Replace(s, " ");
        s = Hyphen.Replace(s, "");
        s = s.Replace('â€“', '-').Replace('â€”', '-');
        s = Puntos.Replace(s, " ");
        return Espacios.Replace(s, " ").Trim();

    public async Task<EmbeddingResponse> GenerarEmbeddingConsultaAsync(string texto, CancellationToken ct = default)
    {
        if (string.IsNullOrWhiteSpace(texto)) return EmbeddingResponse.Fallo("El texto está vacío");
        var r = await EmbedAsync(new[] { PrefijoConsulta + texto.Trim() }, ct);
        return r[0];
    }

    public async Task<IReadOnlyList<EmbeddingResponse>> GenerarEmbeddingsDocumentosAsync(
        IReadOnlyList<string> textos, CancellationToken ct = default)
    {
        var resultados = new List<EmbeddingResponse>(textos.Count);
        for (int i = 0; i < textos.Count; i += _opt.EmbeddingBatchSize)
        {
            var lote = textos.Skip(i).Take(_opt.EmbeddingBatchSize)
                             .Select(t => PrefijoDocumento + t.Trim()).ToList();
            resultados.AddRange(await EmbedAsync(lote, ct));
        }
        return resultados;
    }

    private async Task<EmbeddingResponse[]> EmbedAsync(IReadOnlyList<string> entradas, CancellationToken ct)
    {
        EmbeddingResponse[] FalloTotal(string msg) =>
            Enumerable.Range(0, entradas.Count).Select(_ => EmbeddingResponse.Fallo(msg)).ToArray();

        try
        {
            using var resp = await _http.PostAsJsonAsync(
                "api/embed", new { model = _opt.EmbeddingModel, input = entradas, truncate = true, keep_alive = _opt.KeepAlive }, ct);

            if (!resp.IsSuccessStatusCode)
            {
                var cuerpo = await resp.Content.ReadAsStringAsync(ct);
                return FalloTotal($"Ollama respondió {(int)resp.StatusCode}: {cuerpo}");
            }

            await using var stream = await resp.Content.ReadAsStreamAsync(ct);
            using var doc = await JsonDocument.ParseAsync(stream, cancellationToken: ct);

            if (!doc.RootElement.TryGetProperty("embeddings", out var arr))
                return FalloTotal("La respuesta de Ollama no trae 'embeddings' (¿versión < 0.1.44?)");

            var lista = new List<EmbeddingResponse>();
            foreach (var e in arr.EnumerateArray())
            {
                var v = new float[e.GetArrayLength()];
                int k = 0;
                foreach (var x in e.EnumerateArray()) v[k++] = x.GetSingle();
                lista.Add(v.Length == 0 ? EmbeddingResponse.Fallo("Vector vacío") : EmbeddingResponse.Ok(v));
            }

            return lista.Count == entradas.Count
                ? lista.ToArray()
                : FalloTotal($"Se esperaban {entradas.Count} embeddings y llegaron {lista.Count}");
        }
        catch (OperationCanceledException) when (ct.IsCancellationRequested) { throw; }
        catch (HttpRequestException ex)
        {
            _log.LogError(ex, "No se pudo conectar con Ollama");
            return FalloTotal($"No se pudo conectar con Ollama. ¿Está corriendo? {ex.Message}");
        }
        catch (Exception ex)
        {
            _log.LogError(ex, "Error generando embeddings");
            return FalloTotal(ex.Message);
        }

///    así que cada pregunta costaba dos generaciones (una se tiraba). Con un modelo de 7B eso duplicaba la espera.
///  - Elige el modelo según ChatPrompt.ModoModelo ("rapido" | "preciso"); antes el modo se leía pero nadie lo usaba.
///  - Envía num_ctx, num_predict, temperature, top_p y repeat_penalty de OllamaOptions (antes se ignoraban).

{
    private readonly HttpClient _http;
    private readonly OllamaOptions _opt;
    private readonly ILogger<OllamaChatService> _log;

    public OllamaChatService(IHttpClientFactory factory, IOptions<OllamaOptions> options, ILogger<OllamaChatService> log)
    {
        _http = factory.CreateClient("ollama-chat");
        _opt = options.Value;
        _log = log;
    }

    private string ElegirModelo(string? modo)
    {
        var modelo = modo switch
        {
            "rapido" => _opt.ChatModelRapido,
            "preciso" => _opt.ChatModelPreciso,
            _ => _opt.ChatModel
        };
        return string.IsNullOrWhiteSpace(modelo) ? _opt.ChatModel : modelo;
    }

    public async Task<ChatCompletionResult> GenerarAsync(ChatPrompt prompt, CancellationToken ct = default)
    {
        if (string.IsNullOrWhiteSpace(prompt.Usuario))
            return new ChatCompletionResult { Exito = false, ErrorMensaje = "El prompt de usuario está vacío." };

        var modelo = ElegirModelo(prompt.ModoModelo);

        try
        {
            var cuerpo = new
            {
                model = modelo,
                stream = false,
                keep_alive = _opt.KeepAlive,
                options = new
                {
                    temperature = _opt.Temperature,
                    top_p = _opt.TopP,
                    repeat_penalty = _opt.RepeatPenalty,
                    num_ctx = _opt.NumCtx,
                    num_predict = _opt.NumPredict
                },
                messages = new[]
                {
                    new { role = "system", content = prompt.Sistema },
                    new { role = "user", content = prompt.Usuario }
                }
            };

            var reloj = Stopwatch.StartNew();
            using var res = await _http.PostAsJsonAsync("api/chat", cuerpo, ct);
            var json = await res.Content.ReadAsStringAsync(ct);

            if (!res.IsSuccessStatusCode)
            {
                _log.LogError("Ollama respondió {Status} con el modelo {Modelo}: {Cuerpo}", (int)res.StatusCode, modelo, json);
                return new ChatCompletionResult
                {
                    Exito = false,
                    ErrorMensaje = res.StatusCode == HttpStatusCode.NotFound
                        ? $"El modelo '{modelo}' no está descargado. Ejecuta: ollama pull {modelo}"
                        : $"Ollama respondió {(int)res.StatusCode}."
                };
            }

            using var doc = JsonDocument.Parse(json);
            var raiz = doc.RootElement;
            var texto = raiz.GetProperty("message").GetProperty("content").GetString()?.Trim() ?? "";

            long Ns(string n) => raiz.TryGetProperty(n, out var v) && v.TryGetInt64(out var x) ? x : 0;
            int Cuenta(string n) => raiz.TryGetProperty(n, out var v) && v.TryGetInt32(out var x) ? x : 0;

            int entrada = Cuenta("prompt_eval_count");
            int salida = Cuenta("eval_count");
            double cargaMs = Ns("load_duration") / 1e6;
            double promptMs = Ns("prompt_eval_duration") / 1e6;
            double genMs = Ns("eval_duration") / 1e6;
            double tps = genMs > 0 ? salida / (genMs / 1000.0) : 0;

            _log.LogInformation(

                "generación {GenMs:F0} ms ({GenTok} tok, {Tps:F1} tok/s)",
                modelo, reloj.ElapsedMilliseconds, cargaMs, promptMs, entrada, genMs, salida, tps);

            if (cargaMs > 2000)
                _log.LogWarning("El modelo {Modelo} tardó {S:F1} s en CARGARSE en memoria (no estaba cargado). " +
                                "Con varios modelos, Ollama descarga uno para cargar otro: revisa 'ollama ps' y la RAM libre.",
                                modelo, cargaMs / 1000);

            // Si el prompt ocupa casi toda la ventana, Ollama recorta el INICIO (se perdería el prompt de sistema).
            if (entrada >= _opt.NumCtx - _opt.NumPredict)
                _log.LogWarning("Prompt de {T} tokens con num_ctx {C}: probable recorte. Baja Rag:ContextoMaxCaracteres o sube Ollama:NumCtx.",
                                entrada, _opt.NumCtx);

            if (string.IsNullOrWhiteSpace(texto))
                return new ChatCompletionResult { Exito = false, ErrorMensaje = "El modelo devolvió una respuesta vacía." };

            return new ChatCompletionResult
            {
                Exito = true,
                Respuesta = texto,
                TokensEntrada = entrada,
                TokensSalida = salida
            };
        }
        catch (OperationCanceledException) when (ct.IsCancellationRequested)
        {
            throw;
        }
        catch (TaskCanceledException)
        {
            _log.LogError("Ollama ({Modelo}) no respondió en {S} s (Ollama:TimeoutSeconds)", modelo, _opt.TimeoutSeconds);
            return new ChatCompletionResult
            {
                Exito = false,
                ErrorMensaje = $"Ollama no respondió en {_opt.TimeoutSeconds} s con el modelo {modelo}. ¿Es muy pesado para este equipo?"
            };
        }
        catch (Exception ex)
        {
            _log.LogError(ex, "Error consultando Ollama ({Modelo})", modelo);
            return new ChatCompletionResult { Exito = false, ErrorMensaje = ex.Message };
        }
    }

///  - Expansión de vecinos (parent-child sin cambiar el esquema): el acierto se localiza con el chunk

{

    private readonly IVectorSearchService _search;
    private readonly IVectorIndex _index;
    private readonly RagOptions _opt;
    private readonly ILogger<ContextRetriever> _log;

    public ContextRetriever(IEmbeddingService embeddings, IVectorSearchService search, IVectorIndex index,
        IOptions<RagOptions> opt, ILogger<ContextRetriever> log)
    {
        _embeddings = embeddings; _search = search; _index = index; _opt = opt.Value; _log = log;

    public async Task<IReadOnlyList<ContextoRecuperado>> ObtenerContextoAsync(
        string pregunta, CancellationToken ct = default)
    {
        if (string.IsNullOrWhiteSpace(pregunta)) return Array.Empty<ContextoRecuperado>();

        float[]? vector = null;
        var emb = await _embeddings.GenerarEmbeddingConsultaAsync(pregunta, ct);
        if (emb.Exito) vector = emb.Vector;
        else _log.LogWarning("Embedding de la pregunta falló ({E}); se usa solo búsqueda por palabras clave", emb.ErrorMensaje);

        var aciertos = await _search.BuscarAsync(pregunta, vector, _opt.TopK, ct);
        if (aciertos.Count == 0) return Array.Empty<ContextoRecuperado>();

        var snap = await _index.ObtenerAsync(ct);
        var cubiertos = new HashSet<(string, int)>();
        var resultado = new List<ContextoRecuperado>();

        foreach (var h in aciertos)
        {
            if (cubiertos.Contains((h.Archivo, h.Indice))) continue;

            // NUEVO: Solo expandir vecinos si el acierto principal es FUERTE (ej. score > 0.55)
            // Si apenas pasó el umbral, es mejor enviar solo ese chunk para no meter ruido.
            bool expandirVecinos = h.Score >= 0.55;

            var vecinos = (expandirVecinos && snap.PorArchivo.TryGetValue(h.Archivo, out var doc))
                ? doc.Where(c => Math.Abs(c.Indice - h.Indice) <= _opt.VecinosContexto && c.Seccion == h.Seccion).ToList()
                : new List<IndexedChunk>();

            var vistos = new HashSet<string>();
            var parrafos = new List<string>();
            foreach (var v in vecinos.Count > 0 ? vecinos.Select(x => x.Contenido) : new[] { h.Contenido })
                foreach (var p in v.Split("\n\n", StringSplitOptions.RemoveEmptyEntries))
                    if (vistos.Add(p)) parrafos.Add(p);

            foreach (var v in vecinos) cubiertos.Add((v.Archivo, v.Indice));
            cubiertos.Add((h.Archivo, h.Indice));

            resultado.Add(new ContextoRecuperado(
                string.Join("\n\n", parrafos), h.Titulo, h.Seccion, h.Pagina, h.Score));
        }
        return resultado;
    }

/// <summary>
/// MEJORAS: se normaliza una sola vez (al cargar el Ã­ndice) y luego el coseno es solo
/// producto punto, vectorizado con SIMD. Es stateless â†’ singleton.
/// </summary>
public sealed class SimilarityService
{
    public float[] Normalizar(float[] v)
    {
        double suma = 0;
        foreach (var x in v) suma += (double)x * x;
        var norma = (float)Math.Sqrt(suma);
        var r = new float[v.Length];
        if (norma == 0) { Array.Copy(v, r, v.Length); return r; }
        for (int i = 0; i < v.Length; i++) r[i] = v[i] / norma;
        return r;
    }

    /// <summary>Con vectores ya normalizados, esto ES el coseno.</summary>
    public float ProductoPunto(float[] a, float[] b)
    {
        int n = Math.Min(a.Length, b.Length), i = 0;
        var acc = Vector<float>.Zero;
        int w = Vector<float>.Count;
        for (; i <= n - w; i += w) acc += new Vector<float>(a, i) * new Vector<float>(b, i);
        float suma = Vector.Sum(acc);
        for (; i < n; i++) suma += a[i] * b[i];
        return suma;
    }

    public double CalcularCosineSimilarity(float[] a, float[] b) =>
        a.Length != b.Length ? 0 : ProductoPunto(Normalizar(a), Normalizar(b));
}

    /// BM25 solo sobre la lista de candidatos (p. ej. ya filtrada por cultivo).
    /// </summary>
    private static double[] Bm25Sobre(
        IReadOnlyList<IndexedChunk> chunks,
        IReadOnlyDictionary<string, int> dfGlobal,
        int totalGlobal,
        double longitudPromedio,
        IReadOnlyList<string> terminos)
    {
        var scores = new double[chunks.Count];
        if (terminos.Count == 0) return scores;

        const double k1 = 1.2, b = 0.75;

        foreach (var t in terminos)
        {
            // df y N tienen que ser del MISMO corpus (el Ã­ndice completo). Antes N era el tamaÃ±o del subconjunto
            // filtrado por cultivo: con df global 200 y subconjunto de 50, N = max(50, 200) = 200 y el IDF
            // salÃ­a ~0.0025 (el tÃ©rmino valÃ­a casi cero) en vez de ~0.69 con N = 400.
            if (!dfGlobal.TryGetValue(t, out var df) || df <= 0) continue;
            double N = Math.Max(totalGlobal, df);
            double idf = Math.Log(1 + (N - df + 0.5) / (df + 0.5));

            for (int i = 0; i < scores.Length; i++)
            {
                var ch = chunks[i];
                if (!ch.Tf.TryGetValue(t, out var tf)) continue;
                double norm = tf + k1 * (1 - b + b * ch.LongitudTokens / Math.Max(1, longitudPromedio));
                scores[i] += idf * tf * (k1 + 1) / norm;
            }
        }
        return scores;
    }

        using var suave = new Mat();
        Cv2.GaussianBlur(imagenBgr, suave, new Size(5, 5), 0);

        using var hsv = new Mat();
        Cv2.CvtColor(suave, hsv, ColorConversionCodes.BGR2HSV);

        using var verde = new Mat();
        Cv2.InRange(hsv, VerdeMin, VerdeMax, verde);

        using var afectadoBruto = new Mat();
        Cv2.InRange(hsv, AfectadoMin, AfectadoMax, afectadoBruto);

        using var zonaHoja = new Mat();
        Cv2.Dilate(verde, zonaHoja, nucleoHoja);
