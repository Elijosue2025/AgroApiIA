using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

using System.Security.Cryptography;
using AgroGuia.Aplicacion.DTO.DTOS.Diagnostico;
using AgroGuia.Aplicacion.Servicio;

namespace AgroGuia.Infraestructura.AccesoDatos.Almacen;

public sealed class AlmacenFotosDiagnostico : IAlmacenFotosDiagnostico
{
    private readonly string _raiz;

    public AlmacenFotosDiagnostico(string raizFisica)
    {
        _raiz = Path.GetFullPath(raizFisica);
    }

    public async Task<FotoDiagnosticoGuardadaDto> GuardarAsync(
        long usuarioId, byte[] imagen, CancellationToken ct = default)
    {
        if (usuarioId <= 0)
            throw new ArgumentException("Usuario inválido.", nameof(usuarioId));
        if (imagen is not { Length: > 0 })
            throw new ArgumentException("La imagen está vacía.", nameof(imagen));

        var hash = Convert.ToHexString(SHA256.HashData(imagen)).ToLowerInvariant();
        var nombre = hash + ".jpg";
        var dir = Path.Combine(_raiz, usuarioId.ToString());
        Directory.CreateDirectory(dir);

        var abs = Path.Combine(dir, nombre);
        if (!File.Exists(abs))
            await File.WriteAllBytesAsync(abs, imagen, ct);

        return new FotoDiagnosticoGuardadaDto
        {
            RutaRelativa = $"diagnosticos/{usuarioId}/{nombre}",
            Hash = hash,
            NombreArchivo = nombre
        };
    }

    public async Task<byte[]?> LeerAsync(string rutaRelativa, CancellationToken ct = default)
    {
        var abs = Resolver(rutaRelativa);
        if (abs is null || !File.Exists(abs)) return null;
        return await File.ReadAllBytesAsync(abs, ct);
    }

    private string? Resolver(string rutaRelativa)
    {
        if (string.IsNullOrWhiteSpace(rutaRelativa)) return null;

        var rel = rutaRelativa.Replace('\\', '/').TrimStart('/');
        const string prefijo = "diagnosticos/";
        if (!rel.StartsWith(prefijo, StringComparison.OrdinalIgnoreCase)) return null;

        var abs = Path.GetFullPath(Path.Combine(_raiz, rel[prefijo.Length..]));
        var raiz = _raiz.TrimEnd(Path.DirectorySeparatorChar) + Path.DirectorySeparatorChar;
        return abs.StartsWith(raiz, StringComparison.OrdinalIgnoreCase) ? abs : null;
    }
}

    {
        if (string.IsNullOrWhiteSpace(entidad.Titulo))
            entidad.Titulo = "Nueva Consulta Agronómica";

        if (entidad.FechaCreacion == default)
            entidad.FechaCreacion = DateTime.UtcNow;

        entidad.FechaActualizacion = DateTime.UtcNow;
        entidad.Activo = true;

        await _context.Conversaciones.AddAsync(entidad);
        await _context.SaveChangesAsync();
        return entidad;
    }

    {
        return await _context.Conversaciones
            .FirstOrDefaultAsync(x => x.Id == id);
    }

        long usuarioId, bool soloActivas = true)
    {
        var q = _context.Conversaciones
            .AsNoTracking()
            .Where(x => x.UsuarioId == usuarioId);

        if (soloActivas)
            q = q.Where(x => x.Activo == true);

        return await q
            .OrderByDescending(x => x.FechaActualizacion)
            .ToListAsync();
    }

        long conversacionId, int take = 200)

        if (take > 500) take = 500;

        return await _context.Mensajes
            .AsNoTracking()
            .Where(m => m.ConversacionId == conversacionId)
            .OrderBy(m => m.Fecha)

        await _context.SaveChangesAsync();
    }

        long? usuarioId,
        DateTime? desde,
        DateTime? hasta,
        string? tituloContiene,
        int take = 50)
    {
        if (take <= 0) take = 50;
        if (take > 100) take = 100;

        var q = _context.Conversaciones.AsNoTracking().AsQueryable();

        if (usuarioId is > 0)
            q = q.Where(x => x.UsuarioId == usuarioId.Value);

        if (desde.HasValue)

        if (hasta.HasValue)

        if (!string.IsNullOrWhiteSpace(tituloContiene))
            q = q.Where(x => x.Titulo != null && x.Titulo.Contains(tituloContiene));

        return await q
            .OrderByDescending(x => x.FechaActualizacion)
            .Take(take)
            .ToListAsync();

using AgroGuia.Dominio.Modelo.Abstracciones;
using AgroGuia.Dominio.Modelo.Entidades;
using Microsoft.EntityFrameworkCore;

namespace AgroGuia.Infraestructura.AccesoDatos.Repositorio
{
    public class RepositorioImpl<T> : IRepositorio<T> where T : class
    {
        protected readonly AgroGuiaIA_DBContext _context;
        protected readonly DbSet<T> _dbSet;

        public RepositorioImpl(AgroGuiaIA_DBContext context)
        {
            _context = context;
            _dbSet = context.Set<T>();
        }

        public async Task<IEnumerable<T>> GetAllAsync()
        {
            return await _dbSet.ToListAsync();
        }

        public async Task<T> GetByIdAsync(long id)
        {
            // FindAsync puede devolver null, pero la interfaz espera Task<T> (no Task<T?>).
            // Por lo tanto, lanzamos una excepción si no se encuentra la entidad.
            var entidad = await _dbSet.FindAsync(id);
            if (entidad == null)
                throw new InvalidOperationException($"Entidad de tipo {typeof(T).Name} con id {id} no encontrada.");
            return entidad;
        }

        public async Task AddAsync(T entidad)
        {
            await _dbSet.AddAsync(entidad);
            await _context.SaveChangesAsync();
        }

        public async Task UpdateAsync(T entidad)
        {
            _dbSet.Update(entidad);
            await _context.SaveChangesAsync();
        }

        public async Task DeleteAsync(long id)
        {
            var entidad = await GetByIdAsync(id);

            if (entidad != null)
            {

using AgroGuia.Dominio.Modelo.Abstracciones;
using AgroGuia.Dominio.Modelo.Entidades;
using Microsoft.EntityFrameworkCore;

namespace AgroGuia.Infraestructura.AccesoDatos.Repositorio;

public class VisionIncidenciaRepositorioImpl : IVisionIncidenciaRepositorio
{
    private readonly AgroGuiaIA_DBContext _ctx;

    public VisionIncidenciaRepositorioImpl(AgroGuiaIA_DBContext ctx)
    {
        _ctx = ctx;
    }

    public async Task<VisionIncidencias> CrearAsync(VisionIncidencias entidad, CancellationToken ct = default)
    {
        entidad.FechaCreacion = DateTime.UtcNow;
        entidad.Activo = true;
        await _ctx.VisionIncidencias.AddAsync(entidad, ct);
        await _ctx.SaveChangesAsync(ct);
        return entidad;
    }

    public Task<VisionIncidencias?> ObtenerPorIdAsync(long id, CancellationToken ct = default)
        => _ctx.VisionIncidencias.AsNoTracking()
            .FirstOrDefaultAsync(x => x.Id == id && x.Activo, ct);

    public async Task<IReadOnlyList<VisionIncidencias>> ListarPorUsuarioAsync(
        long usuarioId, int take = 50, CancellationToken ct = default)
    {
        take = take <= 0 ? 50 : Math.Min(take, 200);

        return await _ctx.VisionIncidencias.AsNoTracking()
            .Where(x => x.UsuarioId == usuarioId && x.Activo)
            .OrderByDescending(x => x.FechaCreacion)
            .Take(take)
            .ToListAsync(ct);
    }
}
